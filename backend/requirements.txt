# AWS Lambda Python dependencies
# Package with: pip install -r requirements.txt -t package/
#
# Keep this list to what the package actually imports — everything here is
# unzipped into the Lambda (250 MB cap) and paid for again in cold-start import
# time. boto3 is provided by the python3.11 runtime — do not vendor it.

fastapi==0.111.0
mangum==0.17.0
pydantic==2.7.4

# Pipeline: log template mining (drain3), service graph (neo4j)
drain3==0.9.11
neo4j==5.22.0

# Auth. [crypto] pulls in `cryptography`; without it PyJWT supports HMAC only and
# every RS256-signed Cognito token silently decodes to None — login succeeds and
# every request after it 401s.
PyJWT[crypto]==2.8.0

# Event spine + database monitoring
kafka-python==2.0.2
psycopg2-binary==2.9.9
PyMySQL==1.1.1

# ELK
elasticsearch==8.13.1
python-dateutil==2.8.2

# OTLP/protobuf ingest. Every OpenTelemetry SDK exporter defaults to protobuf, so
# without these the standard onboarding path 400s until the operator finds the one line
# of collector YAML that switches it to JSON. ~1.5MB unzipped; imported lazily in
# shared/elk/otlp.py so a JSON-only deployment pays nothing at cold start.
protobuf==4.25.3
opentelemetry-proto==1.25.0
# Not imported here, but elasticsearch-py's OTel shim does `from opentelemetry import
# trace` and catches only ModuleNotFoundError. opentelemetry-proto makes `opentelemetry`
# a namespace package containing just `.proto`, so that import raises a plain ImportError
# instead — uncaught, at module scope, which took the whole Lambda down with
# Runtime.ImportModuleError. Shipping the api package puts `trace` where it looks.
opentelemetry-api==1.25.0
